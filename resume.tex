\newenvironment{highlights}{
    \begin{itemize}[
        topsep=0.08 cm,
        parsep=0.08 cm,
        partopsep=0pt,
        itemsep=0.03cm,
        leftmargin=0.27cm
    ]%
    % A list resets \parfillskip, so set this after \begin{itemize}. This lets
    % one-line bullets justify across the full available width.
    \setlength{\parfillskip}{0pt}%
}{
    \end{itemize}
} % new environment for highlights

    \onecolentry
    \def\secondColumn{#2}

}{
    \switchcolumn \raggedleft \secondColumn
    \end{paracol}
    \endonecolentry
} % new environment for two column entries
